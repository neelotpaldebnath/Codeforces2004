import sys
 
def process():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
        
    qrTNum = int(input_data[0])
    idx = 1
    out = []
    
    for _ in range(qrTNum):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        k = int(input_data[idx+2])
        idx += 3
        
        d = y - x
        ans = 0
        
        limit = min(x + k - 1, d)
        for j in range(x, limit + 1):
            ans += d % j
            
        start_val = max(x, d + 1)
        end_val = x + k - 1
        if start_val <= end_val:
            ans += d * (end_val - start_val + 1)
            
        out.append(str(ans))
        
    print('
'.join(out))
 
if __name__ == '__main__':
    process()
