import sys
 
def solve():
    # Read all tokens from standard input for fast I/O processing
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    t = int(input_data[0])
    idx = 1
    out = []
    
    for _ in range(t):
        n = int(input_data[idx])
        idx += 1
        
        b = [int(x) for x in input_data[idx:idx+n]]
        idx += n
        
        # Difference array to mark invalidated islands
        elim = [0] * (n + 2)
        
        for i in range(1, n + 1):
            bi = b[i - 1]
            if bi != -1:
                L = max(1, i - bi + 1)
                R = min(n, i + bi - 1)
                if L <= R:
                    elim[L] += 1
                    elim[R + 1] -= 1
                    
        is_elim = [False] * (n + 1)
        curr = 0
        for i in range(1, n + 1):
            curr += elim[i]
            if curr > 0:
                is_elim[i] = True
                
        # Collect all non-eliminated candidate islands
        C = []
        pos_in_C = [0] * (n + 1)
        for i in range(1, n + 1):
            if not is_elim[i]:
                C.append(i)
                pos_in_C[i] = len(C) - 1
                
        forced = [False] * (n + 1)
        m = len(C)
        has_clause = [False] * m
        possible = True
        
        # Process constraints from non-destroyed distance entries
        for i in range(1, n + 1):
            bi = b[i - 1]
            if bi != -1:
                u = i - bi
                v = i + bi
                
                u_valid = (1 <= u <= n and not is_elim[u])
                v_valid = (1 <= v <= n and not is_elim[v])
                
                if not u_valid and not v_valid:
                    possible = False
                    break
                
                if u == v or (u_valid and not v_valid):
                    forced[u] = True
                elif not u_valid and v_valid:
                    forced[v] = True
                else:
                    p_u = pos_in_C[u]
                    p_v = pos_in_C[v]
                    if p_v != p_u + 1:
                        possible = False
                        break
                    has_clause[p_u] = True
                    
        if not possible or m == 0:
            out.append("-1")
            continue
            
        # dp[j][0/1]: boolean indicating if a valid assignment exists for C[0...j] ending in 0/1
        dp = [[False, False] for _ in range(m)]
        c0 = C[0]
        
        dp[0][0] = not forced[c0]
        dp[0][1] = True
        
        for j in range(1, m):
            cj = C[j]
            
            if not forced[cj]:
                if has_clause[j - 1]:
                    dp[j][0] = dp[j - 1][1]
                else:
                    dp[j][0] = dp[j - 1][0] or dp[j - 1][1]
            else:
                dp[j][0] = False
                
            dp[j][1] = dp[j - 1][0] or dp[j - 1][1]
            
        if not dp[m - 1][0] and not dp[m - 1][1]:
            out.append("-1")
            continue
            
        any_forced = any(forced[c] for c in C)
        any_clause = any(has_clause)
        
        x = [0] * m
        
        # If no forced values or clauses exist, pick candidate 0 to guarantee at least one treasure
        if not any_forced and not any_clause:
            x[0] = 1
        else:
            # Reconstruct assignment from DP table
            curr_val = 1 if dp[m - 1][1] else 0
            x[m - 1] = curr_val
            
            for j in range(m - 1, 0, -1):
                if curr_val == 0:
                    if has_clause[j - 1]:
                        prev_val = 1
                    else:
                        prev_val = 0 if dp[j - 1][0] else 1
                else:
                    prev_val = 0 if dp[j - 1][0] else 1
                
                x[j - 1] = prev_val
                curr_val = prev_val
                
        ans = ['0'] * (n + 1)
        for j in range(m):
            if x[j] == 1:
                ans[C[j]] = '1'
                
        out.append("".join(ans[1:]))
        
    treasure_map_fin = "
".join(out)
    print(treasure_map_fin)
 
if __name__ == '__main__':
    solve()
