import sys
 
input = sys.stdin.readline
 
t = int(input())
 
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
 
    diff = [0] * (n + 1)
 
    for k in range(1, n + 1):
        m = a[k - 1]
 
        l = m * k
        r = (m + 1) * k - 1
 
        if l < n:
            diff[l] += 1
            end = min(n, r + 1)
            diff[end] -= 1
 
    cur = 0
    ans = []
 
    for x in range(n):
        cur += diff[x]
        if cur == 0:
            ans.append(x)
 
    print(len(ans))
    print(*ans)
