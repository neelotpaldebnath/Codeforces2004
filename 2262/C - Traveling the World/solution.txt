import sys
 
MOD = 10**9 + 7
MAXN = 200000
 
fact = [1] * (MAXN + 1)
 
for i in range(1, MAXN + 1):
    fact[i] = fact[i - 1] * i % MOD
 
 
def solve():
    input = sys.stdin.readline
 
    t = int(input())
 
    for _ in range(t):
        n = int(input())
        a = list(map(int, input().split()))
 
        k = n // 2
        A = a[-1]
 
        ways = fact[k] * fact[n - k - 2] % MOD
 
        candidates = {A - a[-2], A - a[-3]}
 
        ans = 0
 
        for x in candidates:
            b = []
 
            for i in range(k):
                b.append(i * x)
 
            for i in range(n - k):
                b.append(A - i * x)
 
            b.sort()
 
            if b == a:
                ans = (ans + ways) % MOD
 
        print(ans)
 
 
solve()
