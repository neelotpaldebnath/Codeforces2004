import sys
 
def compute():
    # Read all tokens from standard input for fast I/O processing
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    t = int(input_data[0])
    idx = 1
    MOD = 10**9 + 7
    
    out = []
    
    for _ in range(t):
        n = int(input_data[idx])
        idx += 1
        b = []
        for _ in range(n):
            b.append(int(input_data[idx]))
            idx += 1
        
        # elim array is used as a difference array to mark invalidated islands
        elim = [0] * (n + 2)
        
        for i in range(1, n + 1):
            bi = b[i - 1]
            if bi != -1:
                L = max(1, i - bi + 1)
                R = min(n, i + bi - 1)
                if L <= R:
                    elim[L] += 1
                    elim[R + 1] -= 1
                    
        is_elim = [False] * (n + 1)
        curr = 0
        for i in range(1, n + 1):
            curr += elim[i]
            if curr > 0:
                is_elim[i] = True
                
        # C collects all valid candidate islands
        C = []
        pos_in_C = [0] * (n + 1)
        for i in range(1, n + 1):
            if not is_elim[i]:
                C.append(i)
                pos_in_C[i] = len(C) - 1
                
        forced = [False] * (n + 1)
        has_clause = [False] * len(C)
        possible = True
        
        # Translate the constraints into logical clauses and forced inclusions
        for i in range(1, n + 1):
            bi = b[i - 1]
            if bi != -1:
                u = i - bi
                v = i + bi
                
                u_valid = (1 <= u <= n and not is_elim[u])
                v_valid = (1 <= v <= n and not is_elim[v])
                
                # If both endpoints are invalid, condition cannot be satisfied
                if not u_valid and not v_valid:
                    possible = False
                    break
                    
                if u == v:
                    forced[u] = True
                elif u_valid and not v_valid:
                    forced[u] = True
                elif not u_valid and v_valid:
                    forced[v] = True
                else:
                    # Both are valid - by logic they must be adjacent candidates
                    p_u = pos_in_C[u]
                    p_v = pos_in_C[v]
                    if p_v != p_u + 1:
                        possible = False
                        break
                    has_clause[p_u] = True
                    
        if not possible or len(C) == 0:
            out.append("0")
            continue
            
        # dp[j][0/1] -> number of valid ways considering candidates C[0...j]
        # such that the j-th candidate has (0) no treasure, or (1) a treasure
        dp = [[0, 0] for _ in range(len(C))]
        c0 = C[0]
        
        dp[0][0] = 0 if forced[c0] else 1
        dp[0][1] = 1
        
        for j in range(1, len(C)):
            cj = C[j]
            
            if has_clause[j - 1]:
                # If there's an OR clause, avoiding treasure here forces a treasure in the prior candidate
                dp[j][0] = dp[j - 1][1]
            else:
                dp[j][0] = (dp[j - 1][0] + dp[j - 1][1]) % MOD
                
            dp[j][1] = (dp[j - 1][0] + dp[j - 1][1]) % MOD
            
            if forced[cj]:
                dp[j][0] = 0
                
        ans = (dp[-1][0] + dp[-1][1]) % MOD
        
        any_forced = any(forced[c] for c in C)
        any_clause = any(has_clause)
        
        # The problem states Bessie buried treasure on at least one island.
        # If there are NO absolute obligations (forced bits) and NO clauses,
        # the all-0 assignment (no treasure) is inherently valid in our DP and must be excluded.
        if not any_forced and not any_clause:
            ans = (ans - 1 + MOD) % MOD
            
        out.append(str(ans))
        
    treasure_map_fin = '
'.join(out)
    print(treasure_map_fin)
 
if __name__ == '__main__':
    compute()
