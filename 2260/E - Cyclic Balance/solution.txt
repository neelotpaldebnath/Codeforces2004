import sys
 
def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    q = int(input_data[1])
    s = input_data[2]
    
    # Prefix sums for '0's and '1's
    p0 = [0] * (n + 1)
    p1 = [0] * (n + 1)
    
    # Prefix sums for block transitions
    t01 = [0] * (n + 1)
    t10 = [0] * (n + 1)
    
    for i in range(1, n + 1):
        p0[i] = p0[i - 1] + (1 if s[i - 1] == '0' else 0)
        p1[i] = p1[i - 1] + (1 if s[i - 1] == '1' else 0)
        
        if i > 1:
            t01[i] = t01[i - 1] + (1 if s[i - 2] == '0' and s[i - 1] == '1' else 0)
            t10[i] = t10[i - 1] + (1 if s[i - 2] == '1' and s[i - 1] == '0' else 0)
            
    idx = 3
    out = []
    
    for _ in range(q):
        l = int(input_data[idx])
        r = int(input_data[idx + 1])
        idx += 2
        
        n0 = p0[r] - p0[l - 1]
        n1 = p1[r] - p1[l - 1]
        m = r - l + 1
        
        if n0 == 0:
            k_min = max(1, (n1 + 1) // 2)
        elif n1 == 0:
            k_min = max(1, (n0 + 1) // 2)
        else:
            # Count linear blocks of 0s and 1s
            u0 = (t10[r] - t10[l]) + (1 if s[l - 1] == '0' else 0)
            u1 = (t01[r] - t01[l]) + (1 if s[l - 1] == '1' else 0)
            
            c_base = min(u0, u1)
            
            # 3k >= m - c_base ensures enough inserted characters to create extra blocks
            k_min = max(
                1,
                (n0 + 1) // 2,
                (n1 + 1) // 2,
                c_base,
                (m - c_base + 2) // 3
            )
            
        cost = 4 * k_min - m
        out.append(str(cost))
        
    print('
'.join(out))
 
if __name__ == '__main__':
    solve()
