import sys
 
def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    qrTNum = int(input_data[0])
    idx = 1
    out = []
    
    for _ in range(qrTNum):
        n = int(input_data[idx])
        idx += 1
        
        odds = 0
        mod0 = 0
        mod2 = 0
        
        for _ in range(n):
            val = int(input_data[idx])
            idx += 1
            
            if val % 2 != 0:
                odds += 1
            elif val % 4 == 0:
                mod0 += 1
            else:
                mod2 += 1
                
        out.append(str(max(odds, mod0, mod2)))
        
    print('
'.join(out))
 
if __name__ == '__main__':
    solve()
