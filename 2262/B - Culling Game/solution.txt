import sys
 
input = sys.stdin.readline
 
class Fenwick:
    def __init__(self, n):
        self.n = n
        self.bit = [0] * (n + 1)
 
    def add(self, i, val):
        i += 1
        while i <= self.n:
            self.bit[i] += val
            i += i & -i
 
    def sum(self, i):
        res = 0
        while i > 0:
            res += self.bit[i]
            i -= i & -i
        return res
 
    def kth(self, k):
        idx = 0
        step = 1 << (self.n.bit_length() - 1)
 
        while step:
            nxt = idx + step
            if nxt <= self.n and self.bit[nxt] < k:
                idx = nxt
                k -= self.bit[nxt]
            step >>= 1
 
        return idx
 
 
def solve():
    n = int(input())
    a = list(map(int, input().split()))
    p = [x - 1 for x in map(int, input().split())]
 
    p.reverse()
 
    skill = Fenwick(n)
    important = Fenwick(n)
 
    ans = [0] * n
    cnt = 0
 
    for i in range(n):
        k = p[i]
 
        left_count = important.sum(k)
 
        if left_count == 0:
            champion = k
            important.add(k, 1)
            cnt += 1
        else:
            champion = important.kth(left_count)
 
            if skill.sum(k) - skill.sum(champion) < a[k]:
                champion = k
                important.add(k, 1)
                cnt += 1
 
        skill.add(k, a[k])
 
        while True:
            right_count = important.sum(k + 1)
 
            if right_count == cnt:
                break
 
            j = important.kth(right_count + 1)
 
            if skill.sum(j) - skill.sum(champion) >= a[j]:
                important.add(j, -1)
                cnt -= 1
            else:
                break
 
        ans[i] = cnt - 1
 
    ans.reverse()
    print(*ans)
 
 
t = int(input())
 
for _ in range(t):
    solve()
