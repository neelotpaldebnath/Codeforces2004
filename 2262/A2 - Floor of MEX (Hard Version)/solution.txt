import sys
 
input = sys.stdin.readline
 
MOD = 1000000007
 
def solve():
    n = int(input())
    a = list(map(int, input().split()))
 
    diff = [0] * (n + 1)
 
    for k, m in enumerate(a, 1):
        l = m * k
        r = (m + 1) * k - 1
 
        if l < n:
            diff[l] += 1
            diff[min(n, r + 1)] -= 1
 
    pref = [0] * (n + 1)
    cur = 0
    allowed_count = 0
 
    for x in range(n):
        cur += diff[x]
        if cur == 0:
            allowed_count += 1
        pref[x + 1] = allowed_count
 
    m = allowed_count
 
    need = [0] * (m + 1)
    max_left = 0
 
    for k, val in enumerate(a, 1):
        for j in range(val):
            L = j * k
            R = min(n - 1, (j + 1) * k - 1)
 
            l = pref[L] + 1
            r = pref[R + 1]
 
            if l > r:
                print(0)
                return
 
            if l > need[r]:
                need[r] = l
 
            if l > max_left:
                max_left = l
 
    dp = [0] * (m + 1)
    prefix = [0] * (m + 1)
 
    expired = 0
 
    for i in range(1, m + 1):
        if need[i - 1] > expired:
            expired = need[i - 1]
 
        if expired == 0:
            ways = prefix[i - 1] + 1
        else:
            ways = prefix[i - 1] - prefix[expired - 1]
 
        ways %= MOD
 
        dp[i] = ways
        prefix[i] = (prefix[i - 1] + ways) % MOD
 
    if max_left == 0:
        print((prefix[m] + 1) % MOD)
    else:
        print((prefix[m] - prefix[max_left - 1]) % MOD)
 
t = int(input())
 
for _ in range(t):
    solve()
