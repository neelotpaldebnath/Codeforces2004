import sys
 
def solve():
    # Read all input from standard input
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    
    if n == 1:
        print(1)
        return
 
    MOD = 998244353
 
    # Precompute factorials and combinations
    fact = [1] * (n + 1)
    for i in range(1, n + 1):
        fact[i] = (fact[i - 1] * i) % MOD
        
    inv = [1] * (n + 1)
    inv[n] = pow(fact[n], MOD - 2, MOD)
    for i in range(n - 1, -1, -1):
        inv[i] = (inv[i + 1] * (i + 1)) % MOD
 
    def nPr(n, r):
        if r < 0 or r > n:
            return 0
        return (fact[n] * inv[n - r]) % MOD
 
    # Gather distinct prime factors for each number up to n
    prime_factors = [[] for _ in range(n + 1)]
    for i in range(2, n + 1):
        if not prime_factors[i]:
            for j in range(i, n + 1, i):
                prime_factors[j].append(i)
 
    # Compute Möbius function mu[x] for M <= n
    mu = [0] * (n + 1)
    mu[1] = 1
    for i in range(2, n + 1):
        if not prime_factors[i]:
            continue
        p_prod = 1
        for p in prime_factors[i]:
            p_prod *= p
        if p_prod == i:
            mu[i] = 1 if len(prime_factors[i]) % 2 == 0 else -1
 
    # Precalculate subset products and their sign contributions per number
    prods = [[] for _ in range(n + 1)]
    for i in range(1, n + 1):
        p_list = prime_factors[i]
        cur_prods = [(1, 1)]
        for p in p_list:
            cur_prods += [(x * p, -s) for x, s in cur_prods]
        prods[i] = cur_prods[1:]
 
    # E_array handles generated edge occurrences for M <= n
    E_array = [0] * (n + 1)
    # E_extra handles M > n. We compress edge count and sign into one int memory space
    E_extra = {}
 
    for j in range(1, n):
        for m1, s1 in prods[j]:
            for m2, s2 in prods[j + 1]:
                m = m1 * m2
                if m <= n:
                    E_array[m] += 1
                else:
                    sgn = s1 * s2
                    if m in E_extra:
                        E_extra[m] += sgn
                    else:
                        E_extra[m] = sgn
 
    ans = 1
 
    # Integrate constraints for M <= n
    for M in range(2, n + 1):
        if mu[M] == 0:
            continue
        
        E = E_array[M]
        
        # Calculate isolated edges vs pair components mathematically
        if M == 2:
            C_2 = 0
        elif M % 2 == 0:
            K = M // 2
            C_2 = ((n - 1) // K + 1) // 2
        else:
            C_2 = 0
            
        C_1 = E - 2 * C_2
        if C_1 < 0: C_1 = 0
            
        W = (pow(2, C_1, MOD) * pow(3, C_2, MOD)) % MOD
        
        k = n // M
        coeff = nPr(n, k)
        
        e_S = (coeff * W) % MOD
        term = (-mu[M] * (e_S - 1)) % MOD
        ans = (ans + term) % MOD
 
    # Integrate constraints for M > n
    for M, val in E_extra.items():
        if val == 0:
            continue
            
        # Decompress accumulated value into edge count (E) and original Sign
        if val > 0:
            E, sgn = val, 1
        else:
            E, sgn = -val, -1
            
        if M % 2 == 0:
            K = M // 2
            if K % 2 == 1:
                C_2 = ((n - 1) // K + 1) // 2
            else:
                C_2 = 0
        else:
            C_2 = 0
            
        C_1 = E - 2 * C_2
        if C_1 < 0: C_1 = 0
            
        W = (pow(2, C_1, MOD) * pow(3, C_2, MOD)) % MOD
        
        # k is essentially 0 so coeff defaults to 1
        e_S = W
        term = (-sgn * (e_S - 1)) % MOD
        ans = (ans + term) % MOD
 
    print((ans % MOD + MOD) % MOD)
 
if __name__ == '__main__':
    solve()
