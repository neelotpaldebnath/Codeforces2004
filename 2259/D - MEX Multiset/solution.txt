import sys
 
def solve():
    # Read all tokens from standard input for fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    t = int(input_data[0])
    idx = 1
    out = []
    
    for _ in range(t):
        n = int(input_data[idx])
        a = [int(x) for x in input_data[idx + 1 : idx + 1 + n]]
        idx += 1 + n
        
        zero_count = a.count(0)
        
        # Case 1: Exactly one 0 makes it impossible to satisfy MEX sum >= 2 * max(MEX)
        if zero_count == 1:
            out.append("NO")
        # Case 2: No 0s -> MEX of all sets is 0 (0 >= 0 holds)
        elif zero_count == 0:
            out.append("YES")
            out.append("A" * n)
        # Case 3: At least two 0s -> MEX(A)=1, MEX(B)=1, MEX(C)=0 (1 + 1 + 0 >= 2 * 1 holds)
        else:
            out.append("YES")
            res = []
            first_zero = True
            for val in a:
                if val == 0:
                    if first_zero:
                        res.append('A')
                        first_zero = False
                    else:
                        res.append('B')
                else:
                    res.append('C')
            out.append("".join(res))
            
    print("
".join(out))
 
if __name__ == '__main__':
    solve()
