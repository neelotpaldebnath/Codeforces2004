import sys
 
def solve():
    input = sys.stdin.readline
    t = int(input())
    out = []
 
    for _ in range(t):
        n = int(input())
        a = [0] + list(map(int, input().split()))
        b = [0] + list(map(int, input().split()))
 
        start = 1
        remaining = 0
        possible = True
 
        for i in range(1, n + 1):
            if a[i] > b[i]:
                possible = False
 
            if a[i] < b[i]:
                if remaining == 0:
                    start = i
                remaining += 1
 
        if not possible:
            out.append("-1")
            continue
 
        if remaining == 0:
            out.append("0 1")
            continue
 
        operations = []
        position = start
 
        while remaining > 0:
            if a[position] < b[position]:
                while (
                    a[position] < b[position]
                    and (
                        a[position] >= position
                        or a[a[position]] == b[a[position]]
                    )
                ):
                    a[position] += 1
                    operations.append(1)
 
                if a[position] == b[position]:
                    remaining -= 1
 
            if remaining == 0 or a[position] == position:
                break
 
            operations.append(2)
            position = a[position]
 
        if remaining > 0:
            out.append("-1")
        else:
            out.append(f"{len(operations)} {start}")
            if operations:
                out.append(" ".join(map(str, operations)))
 
    sys.stdout.write("
".join(out))
 
solve()
