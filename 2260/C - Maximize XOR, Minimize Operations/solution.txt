import sys
 
def process():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
        
    qrTNum = int(input_data[0])
    out = []
    idx = 1
    
    memo = [[[( -2, -2 ) for _ in range(2)] for _ in range(2)] for _ in range(31)]
    visited = [[[0] * 2 for _ in range(2)] for _ in range(31)]
    current_test = 0
    
    for _ in range(qrTNum):
        current_test += 1
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        idx += 2
        
        S = x + y
        
        def dfs(bit, tight, c):
            if bit == -1:
                if c == 0:
                    return (0, 0)
                else:
                    return (-1, -1)
                    
            if visited[bit][tight][c] == current_test:
                return memo[bit][tight][c]
                
            best_xor = -1
            best_x = -1
            
            x_bit = (x >> bit) & 1
            S_bit = (S >> bit) & 1
            
            for xp in range(2):
                if tight and xp > x_bit:
                    continue
                    
                for yp in range(2):
                    v = xp + yp
                    cin = S_bit + 2 * c - v
                    if 0 <= cin <= 1:
                        nxt_tight = 1 if (tight and xp == x_bit) else 0
                        res_xor, res_x = dfs(bit - 1, nxt_tight, cin)
                        
                        if res_xor != -1:
                            cur_xor = res_xor | ((xp ^ yp) << bit)
                            cur_x = res_x | (xp << bit)
                            
                            if cur_xor > best_xor or (cur_xor == best_xor and cur_x > best_x):
                                best_xor = cur_xor
                                best_x = cur_x
                                
            visited[bit][tight][c] = current_test
            memo[bit][tight][c] = (best_xor, best_x)
            return best_xor, best_x
            
        ans_xor, ans_x = dfs(30, 1, 0)
        out.append(f"{ans_xor} {x - ans_x}")
        
    print('
'.join(out))
 
if __name__ == '__main__':
    process()
